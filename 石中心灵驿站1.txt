function pickOffline(text){
  for(const rule of OFFLINE_RULES){
    if(rule.k.some(w => text.includes(w))) return rule.r;
  }
  return OFFLINE_DEFAULT[Math.floor(Math.random() * OFFLINE_DEFAULT.length)];
}

async function typeOut(str, onDelta){
  const chunks = str.match(/[\s\S]{1,2}/g) || [];
  for(const c of chunks){
    onDelta(c);
    await sleep(22);
  }
  return str;
}

async function streamChat(messages, onDelta){
  const cfg = getCfg();
  const url = cfg.baseURL.replace(/\/+$/, '') + '/chat/completions';
  const res = await fetch(url, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': 'Bearer ' + cfg.key,
      'Accept': 'text/event-stream'
    },
    body: JSON.stringify({ model: cfg.model, stream: true, temperature: 0.85, messages })
  });

  if(!res.ok){
    const t = await res.text().catch(() => '');
    throw new Error('HTTP ' + res.status + ' ' + t.slice(0, 100));
  }

  const reader = res.body.getReader();
  const dec = new TextDecoder('utf-8');
  let buf = '';

  while(true){
    const { done, value } = await reader.read();
    if(done) break;
    buf += dec.decode(value, { stream: true });

    const lines = buf.split('\n');
    buf = lines.pop();

    for(const raw of lines){
      const line = raw.trim();
      if(!line || !line.startsWith('data:')) continue;
      const data = line.slice(5).trim();
      if(data === '[DONE]') return;
      try {
        const j = JSON.parse(data);
        const d = (j.choices && j.choices[0] && j.choices[0].delta && j.choices[0].delta.content) || '';
        if(d) onDelta(d);
      } catch(e){ /* 忽略不完整分片 */ }
    }
  }
}

async function handleSend(){
  if(sending) return;
  const input = $('input');
  const text = input.value.trim();
  if(!text){ toast('先写点什么吧～'); return; }

  sending = true;
  const sendBtn = $('sendBtn');
  sendBtn.disabled = true;
  sendBtn.textContent = '小灵正在听…';

  const moodEl = moodBox.querySelector('.active');
  const mood = moodEl ? moodEl.dataset.mood : '';

  addBubble('me', text);
  input.value = '';
  history.push({ role: 'user', content: text });

  const aiWrap = addBubble('ai', '');
  const textEl = aiWrap.querySelector('.text');
  textEl.innerHTML = '<span class="typing"><i></i><i></i><i></i></span>';

  let answer = '';
  let started = false;
  const onDelta = d => {
    if(!started){ textEl.textContent = ''; started = true; }
    answer += d;
    textEl.textContent = answer;
    scrollBottom();
  };

  const cfg = getCfg();
  let offlineUsed = false;

  try {
    if(cfg.provider === 'offline' || !cfg.key || !cfg.baseURL){
      offlineUsed = true;
      const reply = pickOffline(text);
      await typeOut(reply, onDelta);
      answer = reply;
    } else {
      const msgs = [{ role:'system', content: SYSTEM_PROMPT }, ...history.slice(-10)];
      await streamChat(msgs, onDelta);
      if(!answer) throw new Error('未收到内容');
    }
  } catch(err){
    // 自动降级到离线话术
    offlineUsed = true;
    if(!started){ textEl.textContent = ''; }
    answer = '';
    const reply = pickOffline(text);
    await typeOut(reply, onDelta);
    answer = reply;
    toast('AI 接口不可用，已切换到离线模式');
    console.warn('AI 调用失败：', err);
  }

  history.push({ role: 'assistant', content: answer });

  // 保存匿名记录
  const risk = isRisk(text);
  const rec = {
    id: Date.now() + '_' + Math.random().toString(36).slice(2, 7),
    time: new Date().toISOString(),
    mood,
    q: text,
    a: answer,
    risk,
    offline: offlineUsed
  };
  addRecord(rec);

  // 高风险 → 推送提醒
  if(risk && cfg.webhook){
    pushWebhook(cfg.webhook, text, mood);
  }

  // 高风险 → 页面提示
  if(risk){
    const tip = document.createElement('div');
    tip.className = 'card';
    tip.style.borderLeft = '4px solid #ef4444';
    tip.innerHTML = '<b style="color:#dc2626">⚠ 请照顾好自己</b><p class="small muted" style="margin:6px 0 0">如果你正在经历很难熬的时刻，请一定要告诉身边的老师或家人。全国心理援助热线 <b>12356</b>，24 小时都有人接听。</p>';
    chatEl.appendChild(tip);
  }

  sending = false;
  sendBtn.disabled = false;
  sendBtn.textContent = '发送';
  scrollBottom();
}

function pushWebhook(url, text, mood){
  const body = {
    msg_type: 'text',
    content: {
      text: '【石中心灵驿站 · 预警】\n时间：' + fmtTime(new Date().toISOString()) +
            (mood ? '\n标签：' + mood : '') +
            '\n\n匿名倾诉内容：\n' + text +
            '\n\n请及时关注并启动人工干预流程。'
    }
  };
  try {
    fetch(url, {
      method: 'POST',
      mode: 'no-cors',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body)
    }).catch(() => {});
  } catch(e){ /* 忽略 */ }
}

function renderRecords(onlyRisk){
  const all = getRecords();
  $('statTotal').textContent = all.length;
  $('statToday').textContent = all.filter(r => isToday(r.time)).length;
  $('statRisk').textContent = all.filter(r => r.risk).length;

  let recs = onlyRisk ? all.filter(r => r.risk) : all;
  recs = recs.slice().reverse();

  const wrap = $('recordList');
  if(!recs.length){
    wrap.innerHTML = '<div class="card muted" style="text-align:center">暂无记录</div>';
    return;
  }

  wrap.innerHTML = recs.map(r => {
    const tags =
      (r.mood ? '<span class="tag">' + esc(r.mood) + '</span>' : '') +
      (r.risk ? '<span class="tag risk">⚠ 需关注</span>' : '') +
      (r.offline ? '<span class="tag">离线</span>' : '');
    return '<div class="rec' + (r.risk ? ' risk' : '') + '">' +
             '<div class="rec-head"><span>' + fmtTime(r.time) + '</span><span>' + tags + '</span></div>' +
             '<div class="rec-q">' + esc(r.q) + '</div>' +
             '<div class="rec-a">' + esc(r.a) + '</div>' +
           '</div>';
  }).join('');
}
